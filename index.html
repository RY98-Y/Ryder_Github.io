Rotten Tomatoes 電影評分爬蟲

使用 Python 的 requests 與 BeautifulSoup 爬取爛番茄電影評分，並整理成結構化資料。

rotten_tomatoes_scraper.py 
# Rotten Tomatoes 電影評分爬蟲範例
import requests
from bs4 import BeautifulSoup
import pandas as pd

def fetch_movie_score(movie_slug: str) -> dict:
    """從 Rotten Tomatoes 取得電影評分"""
    url = f"https://www.rottentomatoes.com/m/{movie_slug}"
    headers = {
        "User-Agent": "Mozilla/5.0 (Educational Project)"
    }

    response = requests.get(url, headers=headers, timeout=10)
    response.raise_for_status()

    soup = BeautifulSoup(response.text, "html.parser")

    title_el = soup.select_one("h1")
    score_el = soup.select_one("[data-qa='tomatometer']")

    return {
        "title": title_el.get_text(strip=True) if title_el else "N/A",
        "tomatometer": score_el.get_text(strip=True) if score_el else "N/A",
        "url": url,
    }

if __name__ == "__main__":
    movies = ["dune_2021", "spider_man_no_way_home", "everything_everywhere_all_at_once"]
    results = []

    for slug in movies:
        data = fetch_movie_score(slug)
        results.append(data)
        print(f"✓ {data['title']}: {data['tomatometer']}")

    df = pd.DataFrame(results)
    df.to_csv("rotten_tomatoes_scores.csv", index=False)
    print("\n已儲存至 rotten_tomatoes_scores.csv")
